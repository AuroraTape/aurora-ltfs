	messages:table {
		prefix:string { "ACF" }

		ACF0001I:string { "Opening a file medium changer (%s)" }
		ACF0002E:string { "%s is not a library directory (no library.conf) and is not under ALTFS_FILE_CHANGER_ROOT" }
		ACF0003E:string { "Cannot read %s (errno %d)" }
		ACF0004E:string { "Cannot write %s (errno %d)" }
		ACF0005I:string { "Injected fault %s for %s" }
		ACF0006I:string { "File medium changer %s: %u storage, %u import/export, %u drive elements" }
		ACF0007E:string { "Cartridge name %s is too long for a drive file (at most %d characters)" }
		ACF0008I:string { "file medium changer plugin options:\n    (none; the changer is the library directory, or a name or serial under ALTFS_FILE_CHANGER_ROOT)\n" }
		ACF0009I:string { "Element %u has a lasting fault (%s): %s refused" }
	}
}
