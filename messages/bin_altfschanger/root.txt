// Messages for the altfschanger command.
root:table {
	messages:table {
		prefix:string { "ACH" }

		ACH0001I:string { "Starting altfschanger, %s version %s, log level %d" }
		ACH0002E:string { "Cannot load the medium changer plugin %s (%d)" }
		ACH0003E:string { "Cannot open the medium changer %s (%d)" }
		ACH0004E:string { "%s failed (%d)" }
		ACH0005I:string { "No medium changer found" }
		ACH0006E:string { "Unknown command %s" }
		ACH0007E:string { "Command %s needs %d argument(s)" }
		ACH0008E:string { "Invalid element address %s" }
		ACH0009E:string { "Invalid element type %s (transport, storage, ie, drive or any)" }
		ACH0010I:string { "Usage: %s [options] <command> [arguments]\n"
			"Options:\n"
			"    -d, --device=<name>       Medium changer; for the file plugin a library directory\n"
			"    -e, --backend=<name>      Medium changer plugin (default: the default changer of altfs.conf)\n"
			"    -i, --config=<file>       Configuration file (default: altfs.conf)\n"
			"    -o <option>               Option passed to the plugin\n"
			"    --verbose=<level>         Log level 0 (errors) to 6\n"
			"    --ready-wait=<seconds>    How long the changer layer waits for a not ready changer (30)\n"
			"    --move-wait=<seconds>     How long a cartridge may be unseen while the picker holds it (5)\n"
			"    -h, --help                This message\n"
			"    -V, --version             Version\n"
			"Commands:\n"
			"    list                      Medium changers this host can reach\n"
			"    info                      Identity, element address assignment and features of the changer\n"
			"    elements [type [start [count]]]\n"
			"                              Element status; type is transport, storage, ie, drive or any\n"
			"    move <source> <dest> [transport]\n"
			"                              MOVE MEDIUM between element addresses\n"
			"    exchange <source> <dest1> <dest2> [transport]\n"
			"                              EXCHANGE MEDIUM\n"
			"    scan [start count]        Make the changer re-read its elements (INITIALIZE ELEMENT STATUS)\n"
			"    tur                       TEST UNIT READY\n"
			"    prevent | allow           Lock or unlock the import/export station\n"
			"    help                      Options of the plugin\n"
			"Commands through the changer layer (cartridges by barcode, drives by serial):\n"
			"    inventory                 The library, its drives and its cartridges\n"
			"    rescan [deep]             Re-read the elements; deep makes the changer re-inventory first\n"
			"    load <barcode> <drive>    Move the cartridge into the drive\n"
			"    return <drive>            Move the cartridge in the drive back to its home slot\n"
			"    swap <barcode> <drive>    Return the cartridge in the drive and load the other one\n"
			"    export <barcode>          Move the cartridge to the import/export station\n"
			"    import <barcode>          Move the cartridge from the import/export station to a slot\n" }
		ACH0011I:string { "%s version %s" }
		ACH0012E:string { "A medium changer must be given with -d" }
		ACH0013E:string { "Option parsing of the plugin failed (%d)" }
		ACH0014W:string { "The path to the medium changer changed during the command, repeating %s" }
		ACH0015E:string { "No default medium changer plugin in the configuration; use -e" }
	}
}
